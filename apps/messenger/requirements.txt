# Рабочие зависимости API. Версии закреплены точно: диапазон означает, что
# образ, собранный сегодня и через неделю из того же коммита, окажется
# разным — и тогда digest перестаёт что-либо гарантировать.
fastapi==0.115.6
uvicorn[standard]==0.34.0
prometheus-client==0.21.1
# Драйвер Postgres. asyncpg, а не psycopg: стек асинхронный, и синхронный
# драйвер в обработчике FastAPI блокирует цикл событий целиком.
#
# Пул PgBouncer работает в режиме transaction, поэтому подготовленные
# выражения выключаются на стороне клиента — см. repositories/postgres.py.
asyncpg==0.30.0

# Проверка подписи токена. PyJWT с cryptography, а не самодельный разбор:
# у реалма два ключа (RS256 для подписи и RSA-OAEP для шифрования), их
# надо различать, а ошибка в проверке подписи - это не дефект, а вход
# без пароля.
pyjwt[crypto]==2.10.1
# Клиент HTTP для открытых ключей. Тот же, на котором построен TestClient,
# то есть в образе он всё равно есть.
httpx==0.28.1

# Лимиты частоты. Redis, а не память процесса: лимит, живущий в поде,
# умножается на число реплик, то есть перестаёт быть лимитом.
redis==5.2.1

# Клиент WebSocket для интеграционного теста realtime. Приходит транзитивно
# через uvicorn[standard], но тест в том же образе обязан полагаться на
# объявленную зависимость, а не на соседство пакетов. Версия согласована
# с uvicorn 0.34.0.
websockets==14.1


# Клиент Kafka. aiokafka, а не confluent-kafka: стек асинхронный, а
# библиотека на librdkafka тянет C-зависимость и блокирующие вызовы,
# которые в цикле событий превращаются в остановку всего процесса.
aiokafka==0.12.0

# Экспорт трасс (гейт G2). Три пакета выпускаются одним номером версии
# и обязаны совпадать: SDK, объявленный при API другой версии, ломается
# на несовпадении протокола контекста.
#
# gRPC, а не HTTP: порт уже задан контрактом развёртывания (4317), и он
# же открыт у коллектора. Цена - бинарное колесо grpcio и около 10-15 МБ
# образа; выигрыш - тот же транспорт, что у соседних сервисов.
opentelemetry-api==1.44.0
opentelemetry-sdk==1.44.0
opentelemetry-exporter-otlp-proto-grpc==1.44.0
